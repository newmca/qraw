# Editor UI: docs/design/ui.md. Sources and public headers are
# globbed (D-014) so new files need no edit of this list.
qraw_glob_sources(ui_sources ${CMAKE_CURRENT_SOURCE_DIR})
file(GLOB ui_headers CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/qraw/ui/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/*.h")
list(SORT ui_headers)

add_library(qraw_ui STATIC ${ui_sources} ${ui_headers})
add_library(qraw::ui ALIAS qraw_ui)
target_include_directories(qraw_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(qraw_ui PRIVATE QRAW_APP_ID="${QRAW_APP_ID}")
target_link_libraries(qraw_ui
    PUBLIC qraw_diag qraw_core qraw_commands qraw_format qraw_render Qt6::Widgets Qt6::PrintSupport)
qraw_configure_target(qraw_ui)

# UI data compiled in under :/qraw/<path relative to resources/>:
# shape palettes (M5, generated by tools/palette/) and diagram templates.
set(QRAW_RESOURCE_DIR "${PROJECT_SOURCE_DIR}/resources")
file(GLOB ui_palette_resources CONFIGURE_DEPENDS "${QRAW_RESOURCE_DIR}/palettes/*.json")
file(GLOB_RECURSE ui_template_resources CONFIGURE_DEPENDS "${QRAW_RESOURCE_DIR}/templates/*.xml")
# Window icon fallback when the icon theme has none: the hicolor PNGs (tools/icons).
file(GLOB_RECURSE ui_icon_resources CONFIGURE_DEPENDS "${QRAW_RESOURCE_DIR}/icons/hicolor/*.png")
set(ui_resources ${ui_palette_resources} ${ui_template_resources} ${ui_icon_resources})
list(SORT ui_resources)
if(ui_resources)
    qraw_add_big_resources(qraw_ui qraw_ui_data
        PREFIX "/qraw"
        BASE "${QRAW_RESOURCE_DIR}"
        FILES ${ui_resources})
endif()

# Translations: English source strings live in i18n/qraw_en.ts. Refresh it
# with `cmake --build --preset dev --target update_translations`; the .qm
# files are compiled in under :/qraw/i18n/.
file(GLOB ui_ts_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/i18n/*.ts")
qt_add_translations(qraw_ui
    TS_FILES ${ui_ts_files}
    RESOURCE_PREFIX "/qraw/i18n"
    LUPDATE_OPTIONS -no-obsolete -locations none)
